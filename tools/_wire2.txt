6|/**
7| * Pre-flight module-tile fit guard for the fixed 4-column v2 certificate.
8| *
9| * The v2 module tiles are FIXED geometry — single line, centered, uppercase,
10| * fixed letterSpacing, and a documented hard floor on font size. The renderer
11| * never silently distorts: if the recipient-entered module title/subtitle
12| * cannot fit the tile even at the minimum allowed font, it REJECTS (fail loud)
13| * rather than shrink out of spec, wrap onto a hidden second line, or overflow.
14| *
15| * That "fail loud at render" is correct, but it is poor design to discover it
16| * only when the admin clicks Preview and gets a raw layout-warning dump.
17| *
18| * This guard runs the SAME fit math the renderer uses (fitTextToField + the
19| * tile's own width/letterSpacing/minFontSize from the template) AHEAD of issue,
20| * so the admin gets an actionable, per-tile reason while still on the form —
21| * and issuance for an exact-match template is blocked before it costs a render.
22| */
23|
24|export interface ModuleTileFitIssue {
25|  /** 1-based tile number this module maps to (1..4). */
26|  order: number;
