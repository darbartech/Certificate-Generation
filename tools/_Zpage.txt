21|// Converts a "data:<mime>;base64,<data>" string into a Blob object URL.
22|// Blob URLs don't have the ~2MB character limit that data URIs hit when
23|// used as an <iframe src> in Chrome, which causes large PDFs to silently
24|// fail to render.
25|function base64DataUriToBlobUrl(dataUri: string): string {
26|  const [header, base64] = dataUri.split(",");
27|  const mimeMatch = header.match(/data:(.*);base64/);
28|  const mimeType = mimeMatch?.[1] || "application/pdf";
29|
30|  const byteChars = atob(base64);
31|  const byteNumbers = new Array(byteChars.length);
32|  for (let i = 0; i < byteChars.length; i++) {
33|    byteNumbers[i] = byteChars.charCodeAt(i);
34|  }
35|  const byteArray = new Uint8Array(byteNumbers);
36|  const blob = new Blob([byteArray], { type: mimeType });
37|  return URL.createObjectURL(blob);
