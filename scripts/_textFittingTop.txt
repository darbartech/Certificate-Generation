68: export const estimateTextWidth = (
69:   text: string,
70:   fontSize: number,
71:   weight: string | number = "normal",
72:   letterSpacing: number = 0,
73:   fontName?: string
74: ): number => {
75:   if (!text) return 0;
76:   const charWidth = estimateCharWidth(fontSize, weight, fontName);
77:   let width = 0;
78:   for (const char of text) {
79:     if (char === " ") {
80:       width += charWidth * 0.5;
81:     } else if ("iIl.,;:| ".includes(char)) {
82:       width += charWidth * 0.38;
83:     } else if ("mMWw".includes(char)) {
84:       width += charWidth * 1.28;
85:     } else if ("fjlrtIJ().!?" .includes(char)) {
86:       width += charWidth * 0.7;
87:     } else if ("abcdeghknopqsuvxyzABCDEGHKLNOPQRSTUVXYZ0123456789".includes(char)) {
88:       width += charWidth;
89:     } else {
90:       width += charWidth;
91:     }
92:   }
93:   if (letterSpacing !== 0 && text.length > 1) {
94:     const effectiveSpacing = (letterSpacing / 1000) * fontSize;
95:     width += effectiveSpacing * (text.length - 1);
96:   }
97:   return width;
98: };

100: export const wrapTextToWidth = (
101:   text: string,
102:   maxWidthPt: number,
103:   fontSize: number,
104:   weight: string | number = "normal",
105:   letterSpacing: number = 0,
106:   fontName?: string
107: ): string[] => {
108:   if (!text) return [];
109: 
110:   const words = text.split(/\s+/);
111:   const lines: string[] = [];
112:   let currentLine = "";
113: 
114:   for (const word of words) {
115:     const testLine = currentLine ? `${currentLine} ${word}` : word;
116:     const lineWidth = estimateTextWidth(testLine, fontSize, weight, letterSpacing, fontName);
117:     if (lineWidth <= maxWidthPt) {
118:       currentLine = testLine;
119:     } else {
120:       if (currentLine) lines.push(currentLine);
121:       const wordWidth = estimateTextWidth(word, fontSize, weight, letterSpacing, fontName);
122:       if (wordWidth > maxWidthPt) {
123:         let remaining = word;
124:         while (remaining.length > 0) {
125:           let chunkLength = 1;
126:           while (
127:             chunkLength < remaining.length &&
128:             estimateTextWidth(remaining.slice(0, chunkLength + 1), fontSize, weight, letterSpacing, fontName) <= maxWidthPt
129:           ) {
130:             chunkLength++;
131:           }
132:           lines.push(remaining.slice(0, chunkLength));
133:           remaining = remaining.slice(chunkLength);
134:         }
135:         currentLine = "";
136:       } else {
137:         currentLine = word;
138:       }
139:     }
140:   }
141:   if (currentLine) lines.push(currentLine);
142: 
143:   return lines;
144: };
